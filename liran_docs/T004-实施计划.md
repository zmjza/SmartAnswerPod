```js
import assert from 'node:assert/strict'
import { build } from 'esbuild'
import { chromium } from 'patchright'

const bundled = await build({ entryPoints: ['electron/detect.ts'], bundle: true,
  platform: 'node', format: 'esm', write: false })
const { listCourses } = await import('data:text/javascript;base64,' +
  Buffer.from(bundled.outputFiles[0].text).toString('base64'))
const browser = await chromium.launch({ headless: true })
try {
  const context = await browser.newContext()
  const href = 'https://l.shou.org.cn/study/HomeWorkNew.aspx?courseOpenId=fixture-a&minorcourseopenid=fixture-m'
  await context.route('**/*', async route => {
    const url = new URL(route.request().url())
    if (url.hostname === 'aiedu.shou.org.cn') {
      await route.fulfill({ contentType: 'text/html', body: `
        <nav aria-label="主导航"><button aria-label="学习">学习</button></nav>
        <div aria-label="课程列表"><article data-learning-course-card
          data-learning-course-ref="fixture-ref" data-learning-course-card-active="true">
          <button aria-label="切换课程：示例课程"></button><h3>示例课程</h3>
        </article></div>
        <div aria-label="课程学习空间">
          <div role="region" aria-label="课程任务">
            <div aria-label="示例课程 课程任务"><div role="list" aria-label="示例课程 课程任务">
              <button data-figma-segmented-tab-key="active" aria-pressed="true">进行中</button>
              <div role="listitem" data-task-tree-node-kind="lesson">
                <button aria-label="去完成：在线学习" onclick="throw Error('错按钮')">去完成</button>
                <button aria-label="去完成：完成课程形考作业"
                  onclick="window.open('${href}', '_blank')">去完成</button>
                <button aria-label="我已完成：完成课程形考作业"
                  onclick="throw Error('修改了任务状态')">我已完成</button>
              </div>
            </div></div>
            <a href="https://l.shou.org.cn/study/learnCatalogNew.aspx?courseOpenId=fixture-a&minorCourseOpenId=fixture-m">查看更多课程详情 »</a>
          </div>
        </div>` })
    } else if (url.hostname === 'l.shou.org.cn') {
      await route.fulfill({ contentType: 'text/html', body: '<div id="onlineHomework"></div>' })
    } else {
      await route.abort()
    }
  })
  const page = await context.newPage()
  page.setDefaultTimeout(1000)
  await page.goto('https://aiedu.shou.org.cn/learning')
  const courses = await listCourses(page, 'fixture-student', 'occupied')
  assert.deepEqual(courses, [{ name: '示例课程', href,
    key: 'course:fixture-a:fixture-m' }])
  assert.equal(context.pages().length, 1, '采集临时 popup 必须释放')
} finally {
  await browser.close()
}
```
